#Launch Rviz and spawn arm

cd ~/swarm/swarm_project/source/ros2_ws
source /opt/ros/<your-ros-distro>/setup.bash
colcon build --packages-select mycobot_description mycobot_280pi_camera_moveit2
source install/setup.bash
ros2 launch mycobot_280pi_camera_moveit2 demo.launch.py

#Optional: add camera_info_url:=file:///path/to/wrist_camera.yaml once calibrated

source ~/swarm/swarm_project/source/ros2_ws/install/setup.bash
ros2 launch mycobot_280pi_camera_moveit2 camera.launch.py

#In a new terminal spawn all of the nodes 

source install/setup.bash
ros2 run controller_manager spawner joint_state_broadcaster

ros2 control list_controllers
